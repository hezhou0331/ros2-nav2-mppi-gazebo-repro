cmake_minimum_required(VERSION 3.16)
project(roamerx_offline_demo LANGUAGES CXX)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
set(UPSTREAM "${CMAKE_CURRENT_SOURCE_DIR}/../../third_party/genisom_roamerx_open" CACHE PATH "Unmodified RoamerX source")
set(PLANNER "${UPSTREAM}/src/navigation/src/navigo_navfn_planner")
add_executable(roamerx_offline_demo main.cpp "${PLANNER}/src/navfn.cpp")
target_compile_features(roamerx_offline_demo PRIVATE cxx_std_17)
target_include_directories(roamerx_offline_demo PRIVATE "${PLANNER}/include")
ament_target_dependencies(roamerx_offline_demo rclcpp)
